\section{Two-stage stochastic integer programming model}
\begin{frame}%[allowframebreaks]
	\frametitle{Two-stage stochastic integer programming model}
	\framesubtitle{1st-stage and 2nd-stage decisions}
	\begin{itemize}
		\item<1-> The shipper wants minimizes the expected cost of operations
		\medskip
		\begin{itemize}
			\item<1-> \textbf{composing a portfolio of carrier suppliers}
			\medskip
			\item<1-> \textbf{diversifying the supplier selection} (reduce the risk of a single supplier solution)
		\end{itemize}
		\bigskip
		\item<2-> The problem can be modeled as a \textbf{two-stage problem}
		\medskip
		\begin{itemize}
			\item<2-> selects the suppliers and buys the capacity at \textbf{\color{blue}contract rates} in the \textbf{1st stage} and
			\medskip
			\item<2-> buys at \textbf{\color{purple}spot rates} the required extra-capacity in the \textbf{2nd stage}%, when complete information is available
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}%[allowframebreaks]
	\frametitle{State of the art}
	\begin{itemize}
		\item<1-> Bin Packing for Capacity Planning in Supply Chain and Transportation (\cite{chapterBernard2024}) 
		\begin{itemize}
 %\footfullcite{chapterBernard2024}
        	\item<1-> \cite{crainic2016logistics} considered the problem of packing an uncertain set of items into bins selected in the 1st-stage and additional bins purchased in the 2nd-stage
	        \item<1-> Neglecting uncertainty, tactical planning problems of transportation are effectively modeled and solved as VCSBPP, e.g., last-mile delivery~(\cite{perboli2021new})
		    \item<1-> The same problem setting of~\cite{crainic2016logistics} is considered in~\cite{bruni2023machine}, from the perspective of e-commerce retailers (such as Amazon or Alibaba), and resorting machine learning predictions to match real-time computational requirements
	        \item<1-> In \cite{crainic2024} the focus is on the uncertain capacity in the 2nd-stage of the bins purchased in the 1st-stage from a single supplier; % (uncertainty is due to possible mechanical failures, delays of delivery, etc.).
	        authors define the Stochastic Variable Cost and Size Bin Packing Problem (SVCSBPP) and perform a comprehensive set of tests under different operational and uncertainty scenarios
		\end{itemize}
    	\item<2-> No problem is considering \textbf{the effect of the ``off-line'' agreements in the capacity planning}
	    \item<3-> We model the supplier selection problem as an \textbf{extension of the two-stage modeling of the SVCSBPP} of~(\cite{crainic2024})
%		\item Progressive-hedging appears as the best algorithm for two-stage SVCSBPPs, e.g.,~\cite{bruni2023machine, crainic2016logistics, crainic2024}
	\end{itemize}
\end{frame}

\begin{frame}%[allowframebreaks]
	\frametitle{Two-stage stochastic integer programming model}
	\framesubtitle{Sets and notation}
	\begin{itemize}
		\item<1-> Given
		\begin{description}
			\footnotesize
			\item<1-> [$\mathcal{M}$:] the set of available suppliers
			\medskip
			\item<1-> [${N}$:] \textbf{\color{blue}minimum nr. of suppliers to contract with}
			\medskip
			\item<1-> [$T$ and $ \cal T$:] the bin types available respectively at the 1st-stage and 2nd-stage
			\medskip
			\item<1-> [$\mathcal{J}^t_m$:] the set of bins of type $ t$ that the supplier $ m$ can provide at the 1st-stage
			\medskip
			\item<1-> [$\theta_m$:] \textbf{\color{blue}the reliability of supplier $ m$} (i.e., the percentage of capacity expected to be actually delivered)
			\medskip
			\item<1-> [$R$ and $\nu$:] the average \textbf{\color{blue}minimum reliability} the shipper targets \textbf{\color{blue}for selected suppliers} and the \textbf{\color{blue}minimum percentage of capacity to buy} at 1st-stage \textbf{\color{blue}for each selected supplier} respectively
			\medskip
			\item<1-> [$\Omega$:] the set of 2nd-stage scenarios
			\medskip
			\item<1-> [${\cal K}(\omega)$:] the set of 2nd-stage bins available under scenario $ \omega \in \Omega$
			\medskip
			\item<1-> [$I(\omega)$:] the set of orders of size under scenario $ \omega \in \Omega$, and each order has size $v_i(\omega)$
		\end{description}
	\end{itemize}
\end{frame}

\begin{frame}%[allowframebreaks]
	\frametitle{Two-stage stochastic integer programming model}
	\framesubtitle{1st-stage variables}
	\begin{itemize}
		\item<1-> Given also
		\begin{description}
			\footnotesize
			\item<1-> [$ f_m$ and $ f^t_m$:] respectively the \textbf{\color{blue}cost for contracting supplier $ m$} and the contract rate for bins of type $ t$ from supplier $ m$
			\medskip
			\item<1-> [$c^t_m$ and $ r^t_m$:] respectively the \textbf{\color{blue}cost the shipper faces} and the \textbf{\color{blue}compensation the supplier pays} for lost capacity (we assume $c^t_m \ge r^t_m$)
			\medskip
			\item<1-> [$ V^t_m$, $ V^\tau$ and ${\cal V}_{j}(\omega)$:] respectively the volumes of 1st-stage and 2nd-stage bins according to type $ t$ or $ \tau$ and of the 1st-stage bin $ j$ under scenario $ \omega \in \Omega$
			\medskip
			\item<1-> [$ g^\tau(\omega)$:] \textbf{\color{blue}spot rate} for a bin of type $\tau$ under scenario $ \omega \in \Omega$
		\end{description}
		\bigskip
		\item<2-> \textbf{1st-stage variables}
		\begin{description}
			\footnotesize
			\item<2-> [$y_m$:] a binary variable that takes value 1 if supplier $ m \in \mathcal{M}$ is selected
			\medskip
			\item<2-> [$y^t_{jm}$:] a binary variable that takes value 1 if bin $ j$ of type $ t \in T$ is bought from supplier $ m \in \mathcal{M}$
		\end{description}
		\bigskip
		\item<3-> and \textbf{2nd-stage variables}
		\begin{description}
			\footnotesize
			\item<3-> [$x_{ij}(\omega)$:] a binary variable that takes value 1 if the order $ i$ is to pack into bin $ j$
			\medskip
			\item<3-> [$z^\tau_{k}(\omega)$:] a binary variable that takes value 1 if the 2nd-stage bin $ k$ of type $ \tau \in {\cal T}$ is used
		\end{description}
	\end{itemize}
%\frameskip
\end{frame}

\begin{frame}%[allowframebreaks]
	\frametitle{Two-stage stochastic integer programming model}
	\framesubtitle{1st-stage problem}
    \onslide<1->
    {
    We have
	\bigskip
	{
	\small
	\begin{equation}
		\label{2s:objf}
		\min_{y} \sum\limits_{m \in \cal M} {f_{m}}y_{m} +  \sum\limits_{t \in T} \sum\limits_{m \in {{\cal M}}} \sum\limits_{j \in {\cal J}_{m}^t} f_{m}^t y_{j,m}^t  + {\mathbb{E}}_{\xi}\left[Q \left(y,\xi(\omega)\right)\right] 
	\end{equation}
    }
    }
    \onslide<2->
    {
	\noindent subject to
    }
    {
    \small
	\begin{alignat}{2}
        \onslide<2->
        {
		\sum\limits_{m \in \cal {M}} y_{m}\geq{N}, & \qquad \label{eq3}\\
        }
        \onslide<3->
        {
		\sum\limits_{t \in T} \sum\limits_{j \in {\cal J}^t_{m}} V_{m}^{t} \ y_{j,m}^{t} \geq \nu \sum\limits_{t \in T} \sum\limits_{m \in {\cal M}} \sum\limits_{j \in {\cal J}^t_{m}} V_{m}^{t} \ y_{j,m}^{t}, & \qquad \forall m \in {\cal M} \label{eq3.2}\\
        }
        \onslide<4->
        {
		\sum\limits_{m \in {{\cal M}}} \ \theta_{m} \ y_{m}  \geq {R} \sum\limits_{m \in {\cal M}} \ y_{m}, & \qquad \label{eq4}\\
        }
        \onslide<5->
        {
%		\sum\limits_{t \in T} \sum\limits_{j \in {\cal J}^t_{m}} y_{j,m}^{t} \leq \left( \sum\limits_{t \in T} |{\cal J}^t_{m}| \right) y_{m}, & \qquad \forall m \in \cal M, \label{eq5}\\
		y_{j,m}^{t} \leq y_{m}, & \qquad \forall t \in T, m \in {\cal M}, j \in {\cal J}^t_{m}, \label{eq5}\\
        }
        \onslide<6->
        {        
		y_{j,m}^t \in \{0, 1\}, & \qquad \forall t \in T, m \in {\cal M}, j \in {\cal J}^t_{m}, \label{eq6}\\
		y_{m} \in \{0, 1\}, & \qquad \forall m \in \cal M \label{eq7},
        }
	\end{alignat}
    }
\end{frame}

\begin{frame}%[allowframebreaks]
	\frametitle{Two-stage stochastic integer programming model}
	\framesubtitle{2nd-stage problem}
	where
	{
	\small
    \onslide<1->
    {
	\begin{equation}
		\begin{split}
			Q \left(y,\xi (\omega)\right) = & \min_{z(\omega), x(\omega)} 
			\sum\limits_{\tau \in \cal T}\sum\limits_{k \in \cal K^{\tau}(\omega)} g^{\tau}(\omega) z_k^{\tau}(\omega) \; + \\
			& + \sum\limits_{t \in T} \sum\limits_{m \in {\cal M}} \sum\limits_{j \in {\cal J}^t_{m}} (c_{m}^t-r_{m}^t)(V^{t} - {\cal V}_{j}(\omega)) y_{j,m}^t \label{eq8}
		\end{split}
	\end{equation}
	}
    }
	\noindent {subject to}
    {
    \small
	\begin{alignat}{2}
        \onslide<2->
        {        
		\sum\limits_{j \in \cal J \cup {\cal K} (\omega)} x_{i,j}(\omega) = 1, & \qquad \forall i \in I(\omega), \label{eq9}\\
        }
        \onslide<3->
        {        
		\sum\limits_{i \in I(\omega)} v_i(\omega) x_{i,j}(\omega) \leq {\cal V}_{j}(\omega) y_{j,m}^t, & \qquad \forall \: t \in T, m\in{{\cal M}}, j \in {\cal J}^t_{m}, \label{eq10}\\
        }
        \onslide<4->
        {        
		\sum\limits_{i \in I(\omega)} v_i(\omega) x_{i,k}(\omega) \leq V^\tau z^{\tau}_k(\omega), & \qquad \forall \: \tau \in {\cal T}, k \in {\cal K}^{\tau}(\omega), \label{eq11}\\
        }
        \onslide<5->
        {        
		x_{d,j}(\omega) \in \mathbb{Z}^+, & \qquad \forall \: d \in D, j \in {\cal J}, \label{eq12}\\
		x_{d,k}(\omega) \in \mathbb{Z}^+, & \qquad \forall \: d \in D, k \in \cal K(\omega), \label{eq13}\\
		z_k^{\tau}(\omega) \in \{0, 1\}, & \qquad \forall \: \tau \in {\cal T}, k \in {\cal K}^{\tau}(\omega). \label{eq14}
        }
	\end{alignat}
    }
\end{frame}


\section{Row aggregation lower bound (and heuristic RP solution)}
\begin{frame}%[allowframebreaks]
	\frametitle{Row aggregation of the recourse problem}
	\begin{itemize}
		\item<1-> Given the total volume of orders $\Upsilon (\omega)$ under scenario $ \omega \in \Omega$
		\bigskip
		\item<2-> We can aggregate constraints~\eqref{eq10} and~\eqref{eq11} over bins, bin types and suppliers, and consequently discard constraints~\eqref{eq9}
		\bigskip
		\item<3-> And we obtain recourse problem
		{
			\small
			\begin{equation*}
				Q \left(y,\xi (\omega)\right)
			\end{equation*}
			\noindent {subject to}
			\begin{alignat}{2}
				\Upsilon (\omega) \leq \sum\limits_{\substack{ m \in {\cal M}, \\ t \in T, \\ j \in {\cal J}^t_{m}}} {\cal V}_{j}(\omega) y_{j,m}^t + \sum\limits_{\substack{\tau \in {\cal T}, \\ k \in {\cal K}^{\tau,s}}} V^\tau z^{\tau}_k(\omega), & \qquad \forall \: \omega \in \Omega, \label{eq10lb}\\
				z_k^{\tau}(\omega) \in \{0, 1\}, & \qquad \forall \: \tau \in {\cal T}, k \in {\cal K}^{\tau}(\omega), \label{eq14lb}
			\end{alignat}
		}
	\end{itemize}
\end{frame}

\begin{frame}%[allowframebreaks]
	\frametitle{Row aggregation of the recourse problem}
	\begin{itemize}
		\item<1-> By solving problem
		{
		\small
		\begin{equation*}
		\min_{y} \sum\limits_{m \in \cal M} {f_{m}}y_{m} +  \sum\limits_{t \in T} \sum\limits_{m \in {{\cal M}}} \sum\limits_{j \in {\cal J}_{m}^t} f_{m}^t y_{j,m}^t  + {\mathbb{E}}_{\xi}\left[Q \left(y,\xi(\omega)\right)\right] 
		\end{equation*}
		\noindent {subject to}
		\begin{equation*}
			\eqref{eq3},~\eqref{eq3.2},~\eqref{eq4},~\eqref{eq5},~\eqref{eq6},~\text{and}~\eqref{eq10lb}
		\end{equation*}
		\begin{alignat*}{2}
		y_{j,m}^t \in \{0, 1\}, & \qquad \forall t \in T, m \in {\cal M}, j \in {\cal J}^t_{m}, \\
		y_{m} \in \{0, 1\}, & \qquad \forall m \in \cal M, \\
		z_k^{\tau}(\omega) \in \{0, 1\}, & \qquad \forall \: \tau \in {\cal T}, k \in {\cal K}^{\tau}(\omega),
		\end{alignat*}
		}
        \item<2->{we obtain a \textbf{valid lower bound} of the problem}
%        \begin{itemize}
%		      \item<2-> Note that we discarded all scenario-dependent packing variables $ x_{ij}(\omega)$
%        \end{itemize}
		\bigskip
		\item<3-> The solution of 1st-stage variables $ y_m$ and $y^t_{jm}$ can then be evaluated in the original problem % for a set of scenarios $ \tilde{\Omega} \subset \Omega$
		\begin{itemize}
			\item<3-> This step provides an \textbf{upper bound} for the \textbf{recourse problem} (RP)
		\end{itemize}
	\end{itemize}
\end{frame}
